rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // ログインしているか（メール確認前でもOK）
    function isSignedIn() {
      return request.auth != null;
    }

    // ログインしていて、かつメール確認（学生認証）が済んでいるか。
    // 画面側のチェックだけだと、ブラウザを通さず直接アクセスされた場合に
    // 確認前のアカウントでもデータを読み書きできてしまうため、ルール側でも確認する。
    // テスト用アカウント（採用担当者様向け）だけは例外として許可する。
    function isVerified() {
      return isSignedIn() && (
        request.auth.token.email_verified == true
        || request.auth.token.email == 'test@university.ac.jp'
      );
    }

    function isParticipant(data) {
      return isVerified() && request.auth.uid in data.participants;
    }

    function hasOnlyAllowedUserFields() {
      return request.resource.data.keys().hasOnly([
        'name',
        'school',
        'avatarColor',
        'avatarUrl',
        'totalPoints',
        'matches',
        'blockedUserIds',
        'verified',
        'updatedAt'
      ]);
    }

    function isSkillOwner(data) {
      return isVerified() && data.userId == request.auth.uid;
    }

    function isSkillParticipant(data) {
      return isVerified() && (
        data.teacherId == request.auth.uid ||
        data.learnerId == request.auth.uid
      );
    }

    match /conversations/{convId} {
      allow read: if isParticipant(resource.data);
      allow create: if isVerified()
        && request.resource.data.participants is list
        && request.auth.uid in request.resource.data.participants;
      allow update, delete: if isParticipant(resource.data);

      match /messages/{messageId} {
        allow read, write: if isParticipant(
          get(/databases/$(database)/documents/conversations/$(convId)).data
        );
      }
    }

    match /calls/{convId} {
      allow read, write: if isVerified()
        && (
          request.auth.uid == resource.data.callerId
          || request.auth.uid == resource.data.calleeId
          || request.auth.uid == request.resource.data.callerId
          || request.auth.uid == request.resource.data.calleeId
        );

      match /callerCandidates/{docId} {
        allow read, write: if isVerified()
          && (
            request.auth.uid
              == get(/databases/$(database)/documents/calls/$(convId)).data.callerId
            || request.auth.uid
              == get(/databases/$(database)/documents/calls/$(convId)).data.calleeId
          );
      }
      match /calleeCandidates/{docId} {
        allow read, write: if isVerified()
          && (
            request.auth.uid
              == get(/databases/$(database)/documents/calls/$(convId)).data.callerId
            || request.auth.uid
              == get(/databases/$(database)/documents/calls/$(convId)).data.calleeId
          );
      }
    }

    match /users/{userId} {
      allow read: if isVerified();
      // このアプリはクライアント側トランザクションで相手ユーザーの
      // matches / totalPoints も更新するため、更新者を本人に限定できない。
      // その代わり書き込み可能フィールドを制限する。
      allow create, update: if isVerified() && hasOnlyAllowedUserFields();
      allow delete: if isVerified() && request.auth.uid == userId;
    }

    // 新規登録の直後（メール確認前）に自分の登録情報を書き込むため、ここだけは確認前でも許可する
    match /authUsers/{userId} {
      allow read, write, delete: if isSignedIn() && request.auth.uid == userId;
    }

    match /skills/{skillId} {
      allow read: if isVerified();
      allow create: if isVerified() && isSkillOwner(request.resource.data);
      allow update: if isSkillOwner(resource.data) || isSkillParticipant(resource.data);
      allow delete: if isSkillOwner(resource.data);
    }

    match /blocks/{userId} {
      allow read, write: if isVerified() && request.auth.uid == userId;
    }

    match /reports/{reportId} {
      allow create: if isVerified();
      // 現状は運営UIを実装していないため、作成者本人のみ読める設定にする。
      allow read: if isVerified() && resource.data.reporterId == request.auth.uid;
      allow update, delete: if false;
    }
  }
}
